"""Simple disk cache for LLM responses."""
import hashlib
import json
from pathlib import Path


CACHE_DIR = Path(".cache/understanding")
CACHE_DIR.mkdir(parents=True, exist_ok=True)


def _key(text: str) -> str:
    return hashlib.sha256(text.encode("utf-8")).hexdigest()[:16]


def get(text: str) -> dict | None:
    path = CACHE_DIR / f"{_key(text)}.json"
    if path.exists():
        return json.loads(path.read_text(encoding="utf-8"))
    return None


def set(text: str, data: dict) -> None:
    path = CACHE_DIR / f"{_key(text)}.json"
    path.write_text(json.dumps(data, ensure_ascii=False), encoding="utf-8")

"""Deterministic quality checks for Sahayak action plans."""

import re

from app.schemas.citizen import CitizenProfile
from app.schemas.plan import ActionPlan, CriticVerdict
from app.schemas.research import ResearchResult


def _normalize(text: str) -> str:
    return re.sub(r"\s+", " ", text or "").strip().casefold()


def _check_gender_mismatch(profile, plan):
    errors = []

    if profile.gender != "male":
        return errors

    terms = (
        "widow pension", "widow scheme", "maternity benefit",
        "pregnant woman", "pregnant women", "women-only", "women only",
    )

    for scheme in plan.eligible_schemes:
        text = _normalize(
            f"{scheme.name} {scheme.benefits} "
            f"{scheme.why_relevant} {scheme.eligibility_note}"
        )
        if any(term in text for term in terms):
            errors.append(f"Possible gender mismatch for recommendation: {scheme.name}")

    return errors


def _check_age_mismatch(profile, plan):
    errors = []

    if profile.age is None or profile.age >= 60:
        return errors

    terms = (
        "old age pension", "senior citizen pension",
        "senior citizens pension", "vayoshree", "vayovandana",
    )

    for scheme in plan.eligible_schemes:
        if any(term in _normalize(scheme.name) for term in terms):
            errors.append(f"Check age eligibility for recommendation: {scheme.name}")

    return errors


def _check_remarriage_mismatch(profile, plan):
    errors = []
    events = {str(event).casefold() for event in (profile.life_events or [])}

    if not (
        {"widow", "widower", "spouse_death"} & events
    ) or profile.age is None or profile.age < 50:
        return errors

    for scheme in plan.eligible_schemes:
        text = _normalize(
            f"{scheme.name} {scheme.benefits} {scheme.eligibility_note}"
        )
        if "remarriage" in text and ("widow" in text or "widower" in text):
            errors.append(f"Verify remarriage-related eligibility: {scheme.name}")

    return errors


def _check_completeness(plan):
    errors = []
    warnings = []

    if not plan.eligible_schemes and not plan.next_steps:
        errors.append("The plan has no recommendations and no next steps.")

    # These implementation/schema field names must never appear in
    # citizen-facing next-step text.
    internal_names = (
        "scheme_id",
        "evidence_quote",
        "source_page",
        "retrieved_chunks",
        "estimated_annual_benefit",
        "eligible_schemes",
        "document_checklist",
        "time_estimate",
        "source_edition",
        "source_name",
    )

    for index, step in enumerate(plan.next_steps, start=1):
        normalized = step.casefold()

        if len(step.strip()) < 8:
            warnings.append(f"Next step {index} may be too vague.")

        if any(name in normalized for name in internal_names):
            errors.append(
                f"Next step {index} contains an internal schema field name."
            )

    if plan.eligible_schemes:
        if not plan.document_checklist:
            warnings.append("Recommendations exist, but the document checklist is empty.")
        if not plan.timeline:
            warnings.append("Recommendations exist, but the timeline is empty.")

    return errors, warnings


def _check_evidence(plan, research):
    errors = []
    warnings = []
    sources = {}

    for chunk in research.chunks:
        if chunk.chunk_id:
            sources[chunk.chunk_id] = chunk
        if chunk.doc_id:
            sources[chunk.doc_id] = chunk

    for scheme in plan.eligible_schemes:
        source = sources.get(scheme.scheme_id)

        if source is None:
            errors.append(f"No matching retrieved source for: {scheme.name}")
            continue

        quote = _normalize(scheme.evidence_quote)
        source_text = _normalize(source.text)

        if len(quote) < 20:
            errors.append(f"Evidence quote is missing or too short for: {scheme.name}")
        elif quote not in source_text:
            errors.append(f"Evidence quote does not match retrieved text for: {scheme.name}")

        if not source.source_url:
            warnings.append(f"No source URL is available for: {scheme.name}")

    return errors, warnings


def critique(
    profile: CitizenProfile,
    plan: ActionPlan,
    research: ResearchResult | None = None,
) -> CriticVerdict:
    """Evaluate plan quality and evidence grounding."""
    errors = []
    warnings = []

    if profile.mode == "scheme":
        errors.extend(_check_gender_mismatch(profile, plan))
        errors.extend(_check_age_mismatch(profile, plan))
        errors.extend(_check_remarriage_mismatch(profile, plan))

    completeness_errors, completeness_warnings = _check_completeness(plan)
    errors.extend(completeness_errors)
    warnings.extend(completeness_warnings)

    if research is not None and research.chunks:
        evidence_errors, evidence_warnings = _check_evidence(plan, research)
        errors.extend(evidence_errors)
        warnings.extend(evidence_warnings)

    return CriticVerdict(
        passed=not errors,
        errors=errors,
        warnings=warnings,
    )

def bind_sources(

        if chunk.doc_id:
            sources[chunk.doc_id] = chunk

        source = sources.get(option.scheme_id)

        if source is None:
            option.source_url = ""
            option.source_name = ""
            option.source_page = 0
            option.source_edition = ""
            continue

        option.source_url = source.source_url or ""
        option.source_name = source.name or ""

    if profile is not None and profile.mode == "mentor":

                "आवेदन करने से पहले आधिकारिक संस्थान से वर्तमान आवश्यकताओं "

def _build_user_message(
    profile: CitizenProfile,
    research: ResearchResult,
    max_chunks: int,
) -> str:
    """Build a readable evidence prompt with explicit source IDs."""
    today = datetime.now(ZoneInfo("Asia/Kolkata")).date().isoformat()

    evidence_blocks = []

    for chunk in research.chunks[:max_chunks]:
        source_id = chunk.chunk_id or chunk.doc_id or "unknown"

        evidence_blocks.append(
            f"Source ID: {source_id}\n"
            f"Document ID: {chunk.doc_id or ''}\n"
            f"Name: {chunk.name or ''}\n"
            f"Page: {chunk.page}\n"
            f"Edition: {chunk.edition or ''}\n"
            f"Source URL: {chunk.source_url or ''}\n"
            f"Evidence text:\n{chunk.text}"
        )

    evidence_text = (
        "\n\n---\n\n".join(evidence_blocks)
        if evidence_blocks
        else "No retrieved evidence available."
    )

    return (
        f"Today in India: {today}\n\n"
        f"Citizen profile:\n{profile.model_dump_json(indent=2)}\n\n"
        f"Language: {profile.language}\n\n"
        f"Retrieved evidence:\n{evidence_text}\n\n"
        "Create an action plan using only the supplied evidence. "
        "For each recommendation, copy its exact Source ID into scheme_id. "
        "Evidence quotes must be verbatim text from the matching source. "
        "Do not invent eligibility, benefits, deadlines, or contact details. "
        "Return one valid JSON object matching the ActionPlan structure. "
        "Do not include Markdown fences or explanatory text."
    )


@traceable(name="sahayak_plan")
def plan(
    profile: CitizenProfile,
    research: ResearchResult,
    max_chunks: int = 5,
) -> ActionPlan:
    """Generate, validate, and ground an action plan."""
    if research is None or not research.chunks:
        return _fallback_plan(profile)

    prompt = _get_prompt()
    client = _get_client()

    system_prompt = (
        prompt["system"]
        + "\n\nReturn only one valid JSON object. Follow the ActionPlan "
        "structure. Use empty strings, empty lists, or zero where "
        "information is unavailable. Never invent evidence."
    )

    user_message = _build_user_message(profile, research, max_chunks)
    last_error = None

    for attempt in range(5):
        try:
            response = client.chat.completions.create(
                model=MODEL,
                messages=[
                    {"role": "system", "content": system_prompt},
                    {"role": "user", "content": user_message},
                ],
                temperature=0.2,
                response_format={"type": "json_object"},
            )

            raw_json = response.choices[0].message.content

            if not raw_json or not raw_json.strip():
                last_error = ValueError("Groq returned an empty response.")
                time.sleep(min(attempt + 1, 3))
                continue

            data = json.loads(raw_json)
            action_plan = ActionPlan.model_validate(data)

            return bind_sources(action_plan, research, profile)

        except RateLimitError as exc:
            raise RuntimeError(
                "Groq rate limit reached while generating the action plan. "
                "Please retry later."
            ) from exc

        except (json.JSONDecodeError, ValidationError, ValueError) as exc:
            last_error = exc
            print(
                f"[Plan] Invalid response on attempt {attempt + 1}/5: {exc}"
            )
            time.sleep(min(attempt + 1, 3))

        except Exception as exc:
            last_error = exc
            print(
                f"[Plan] Groq error on attempt {attempt + 1}/5: "
                f"{type(exc).__name__}: {exc}"
            )
            time.sleep(min(attempt + 1, 3))

    raise RuntimeError(
        f"Failed to generate a valid action plan after 5 attempts. "
        f"Last error: {last_error}"
    ) from last_error

"""Revise an action plan using critic feedback and retrieved evidence."""

import json
import os
import sys
import time
from pathlib import Path

import yaml
from dotenv import load_dotenv
from groq import Groq, RateLimitError
from pydantic import ValidationError
from langsmith import traceable

ROOT = Path(__file__).resolve().parents[2]

load_dotenv(ROOT / ".env")

from app.agents.evidence import bind_sources

PROMPT_PATH = ROOT / "app" / "prompts" / "replanner" / "v1.yaml"

_prompt_cache = None
_client_cache = None


def _get_prompt():
    """Load and cache the replanning prompt."""
    global _prompt_cache

    if _prompt_cache is None:
        with PROMPT_PATH.open("r", encoding="utf-8") as file:
            _prompt_cache = yaml.safe_load(file)

    return _prompt_cache


def _get_client():
    """Create and cache the Groq client."""
    global _client_cache

    if _client_cache is None:
        api_key = os.getenv("GROQ_API_KEY")
        if not api_key:
            raise RuntimeError(
                "GROQ_API_KEY is missing. Add it to your .env file."
            )
        _client_cache = Groq(api_key=api_key)

    return _client_cache


def _build_user_message(
    profile: CitizenProfile,
    plan: ActionPlan,
    issues: list[str],
    research: ResearchResult | None,
) -> str:
    """Build the replanning request with available evidence."""
    evidence = []

    if research and research.chunks:
        for chunk in research.chunks:
            evidence.append(
                {
                    "chunk_id": chunk.chunk_id,
                    "doc_id": chunk.doc_id,
                    "name": chunk.name,
                    "page": chunk.page,
                    "source_url": chunk.source_url,
                    "text": chunk.text,
                }
            )

    return (
        f"Citizen profile:\n{profile.model_dump_json(indent=2)}\n\n"
        f"Current action plan:\n{plan.model_dump_json(indent=2)}\n\n"
        f"Critic issues to fix:\n{json.dumps(issues, ensure_ascii=False)}\n\n"
        f"Retrieved evidence:\n"
        f"{json.dumps(evidence, ensure_ascii=False, indent=2)}\n\n"
        "Return one valid JSON object matching the ActionPlan structure. "
        "Fix the listed issues without inventing facts or evidence. "
        "If the evidence does not support a scheme, remove it rather than "
        "making unsupported claims. Preserve useful correct information."
    )


@traceable(name="sahayak_replan")
def replan(
    profile: CitizenProfile,
    plan: ActionPlan,
    issues: list[str],
    max_retries: int = 3,
    research: ResearchResult | None = None,
) -> ActionPlan:
    """Revise a plan and validate the result."""
    if not issues:
        return plan

    if research is None or not research.chunks:
        # Avoid asking the model to invent evidence when none is available.
        return plan

    prompt = _get_prompt()
    client = _get_client()

    system_prompt = (
        prompt["system"]
        + "\n\nReturn only a valid JSON object matching ActionPlan. "
        "Use the provided research as the only evidence for scheme claims."
    )

    user_message = _build_user_message(
        profile,
        plan,
        issues,
        research,
    )

    last_error = None

    for attempt in range(max_retries):
        try:
            response = client.chat.completions.create(
                model=MODEL,
                messages=[
                    {"role": "system", "content": system_prompt},
                    {"role": "user", "content": user_message},
                ],
                temperature=0.2,
                response_format={"type": "json_object"},
            )

            raw_json = response.choices[0].message.content

            if not raw_json or not raw_json.strip():
                last_error = ValueError("Groq returned an empty response.")
                time.sleep(min(attempt + 1, 2))
                continue

            data = json.loads(raw_json)
            revised_plan = ActionPlan.model_validate(data)

            return bind_sources(revised_plan, research, profile)

        except RateLimitError as exc:
            raise RuntimeError(
                "Groq rate limit reached while revising the action plan."
            ) from exc

        except (json.JSONDecodeError, ValidationError, ValueError) as exc:
            last_error = exc
            print(
                f"[Replan] Invalid response on attempt "
                f"{attempt + 1}/{max_retries}: {exc}"
            )
            time.sleep(min(attempt + 1, 2))

        except Exception as exc:
            last_error = exc
            print(
                f"[Replan] Error on attempt "
                f"{attempt + 1}/{max_retries}: "
                f"{type(exc).__name__}: {exc}"
            )
            time.sleep(min(attempt + 1, 2))

    raise RuntimeError(
        f"Failed to revise the action plan. Last error: {last_error}"
    ) from last_error

"""Research Agent — retrieves grounded information with citations."""
import sys
import re
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent.parent))

from app.schemas.citizen import CitizenProfile
from app.schemas.research import ResearchResult, RetrievedChunk
from langsmith import traceable

MODES_WITH_DATA = {"scheme", "mentor"}
MIN_RERANK_SCORE = 0.5
# Cross-encoder logits are not probabilities. Reviewed career catalogues use a
# separate cutoff, checked against supported paraphrases and unrelated controls.
MIN_MENTOR_RERANK_SCORE = -1.0
STATE_BOOST = 5.0           # bigger boost for local schemes


def _build_query(profile: CitizenProfile) -> str:
    """Build a search query from the profile AND raw input."""
    parts = []

    if profile.retrieval_query.strip():
        parts.append(profile.retrieval_query.strip())

    if profile.mode == "scheme":
        if profile.life_events:
            for event in profile.life_events:
                if event == "spouse_death":
                    parts.append("widow pension")
                elif event == "retirement":
                    parts.append("old age pension")
                elif event == "education_milestone":
                    parts.append("scholarship")
        if profile.category:
            parts.append(f"{profile.category} scholarship")
        if profile.age and profile.age >= 60:
            parts.append("senior citizen scheme")
        parts.append("government scheme")

    elif profile.mode == "mentor":
        parts.append("What education and courses lead to " + (profile.career_goal or "a suitable career") + "?")
        if profile.education_level:
            parts.append(profile.education_level)
        parts.extend(profile.subjects)
        parts.extend(profile.interests)
        parts.extend(profile.strengths)
        if profile.career_goal:
            parts.append(profile.career_goal)

    elif profile.mode == "journey":
        parts.append("travel guide route itinerary")

    elif profile.mode == "care":
        parts.append("emotional support helpline")

    elif profile.mode == "companion":
        parts.append("local news")

    # ALWAYS include raw_input words too — profile may be incomplete
    if profile.raw_input:
        parts.append(profile.raw_input)

    # Deduplicate while preserving order
    seen = set()
    unique = []
    for p in parts:
        if p.lower() not in seen:
            seen.add(p.lower())
            unique.append(p)
    return " ".join(unique)

def _source_matches_intent(doc_id: str, query: str) -> bool:
    """Avoid confusing exam bulletins that share school-subject vocabulary.

    Apply only to known, narrowly scoped entrance bulletins and explicit career
    terms. General exploration and comparisons retain both sources.
    """
    medical = bool(re.search(r"\b(doctor|medicine|medical|mbbs|bds|dentist|neet)\b", query, re.I))
    other = bool(re.search(r"\b(engineering|engineer|architecture|architect|jee|b[ .-]?tech|commerce|accounting|accountancy|business|finance|software|computer)\b", query, re.I))
    if doc_id == "nta_neet_2026" and other and not medical:
        return False
    if doc_id == "nta_jee_2026" and medical and not other:
        return False
    return True


@traceable(run_type="chain", name="2. Research Agent")
def research(profile: CitizenProfile, top_k: int = 5) -> ResearchResult:
    """Find grounded information for the given profile."""
    # Lazy imports — heavy libraries only load when actually needed
    from app.rag.embed import embed_text
    from app.rag.vector_store import search as vector_search
    from app.rag.bm25 import get_bm25_index
    from app.rag.rrf import reciprocal_rank_fusion
    from app.rag.rerank import rerank

    query = _build_query(profile)

    if profile.mode not in MODES_WITH_DATA:
        return ResearchResult(
            query=query,
            mode=profile.mode,
            chunks=[],
            num_results=0,
            source="no_data_for_mode",
        )

    # 1. Vector search
    query_vec = embed_text(query)
    vec_results = vector_search(query_vec, n_results=25, mode=profile.mode,
                                state=profile.state if profile.mode == "scheme" else None)

    vec_hits = []
    for i in range(len(vec_results["ids"][0])):
        meta = vec_results["metadatas"][0][i]
        vec_hits.append({
            "id": vec_results["ids"][0][i],
            "text": vec_results["documents"][0][i],
            "name": meta.get("name", ""),
            "doc_id": meta.get("doc_id", ""),
            "state": meta.get("state", ""),
            "mode": meta.get("mode", "scheme"),
            "source_url": meta.get("source_url", ""),
            "page": meta.get("page", 0),
            "edition": meta.get("edition", ""),
            "retrieved_on": meta.get("retrieved_on", ""),
        })

    # 2. BM25 search
    bm25 = get_bm25_index()
    bm25_hits = bm25.search(query, top_k=25, mode=profile.mode,
                             state=profile.state if profile.mode == "scheme" else None)

    # 3. RRF fusion
    fused = reciprocal_rank_fusion([vec_hits, bm25_hits], id_key="id")

    # Exclude clearly mismatched entrance bulletins before reranking.
    if profile.mode == "mentor":
        fused = [c for c in fused if _source_matches_intent(c.get("doc_id", ""), query)]

    # 4. Rerank with source context.
    for candidate in fused:
        candidate["ranking_text"] = candidate.get("name", "") + "\n" + candidate["text"]
    reranked = rerank(query, fused[:35], text_key="ranking_text", top_k=35)

    # 5. Filter + state boost + dedupe
    user_state = (profile.state or "").lower()

    scored = []
    for r in reranked:
        if r.get("mode") != profile.mode:
            continue
        chunk_state = (r.get("state", "") or "").lower()

        # State filter: if user has a state, allow only central + that state
        if profile.mode == "scheme" and user_state and chunk_state and chunk_state not in (user_state, "central"):
            continue

        score = r.get("_rerank_score", 0.0)
        cutoff = MIN_MENTOR_RERANK_SCORE if profile.mode == "mentor" else MIN_RERANK_SCORE
        if score < cutoff:
            continue

        # Boost local schemes
        if profile.mode == "scheme" and user_state and chunk_state == user_state:
            score += STATE_BOOST

        scored.append((r, score))

    # Re-sort by adjusted score
    scored.sort(key=lambda x: x[1], reverse=True)

    # Dedupe by document
    chunks = []
    seen_docs = {}
    for r, adjusted_score in scored:
        doc_id = r.get("doc_id", "")
        if seen_docs.get(doc_id, 0) >= 2:
            continue
        seen_docs[doc_id] = seen_docs.get(doc_id, 0) + 1

        chunks.append(RetrievedChunk(
            chunk_id=r["id"], page=r.get("page", 0), edition=r.get("edition", ""),
            retrieved_on=r.get("retrieved_on", ""),
            doc_id=doc_id,
            name=r.get("name", doc_id),
            text=r["text"],
            rerank_score=adjusted_score,
            source_url=r.get("source_url", ""),
            citation=f"{r.get('name', doc_id)} — {r.get('source_url', 'source not recorded')}",
        ))

        if len(chunks) >= top_k:
            break

    return ResearchResult(
        query=query,
        mode=profile.mode,
        chunks=chunks,
        num_results=len(chunks),
        source="rag",
        diagnostics={"dense_doc_ids": list(dict.fromkeys(c["doc_id"] for c in vec_hits))[:5],
                     "sparse_doc_ids": list(dict.fromkeys(c["doc_id"] for c in bm25_hits))[:5],
                     "fused_doc_ids": list(dict.fromkeys(c["doc_id"] for c in fused))[:5],
                     "dense_candidates": len(vec_hits), "sparse_candidates": len(bm25_hits),
                     "fused_candidates": len(fused), "kept": len(chunks),
                     "top_rerank_score": reranked[0].get("_rerank_score") if reranked else None},
    )

PROMPT_PATH = Path(__file__).resolve().parents[1] / "prompts/understanding/v1.yaml"
MODEL = os.getenv("GROQ_MODEL", "openai/gpt-oss-20b")

VALID_CATEGORIES = {"general", "obc", "sc", "st", "ews"}

GENDER_KEYWORDS = {
    "widower": "male",
    "widow": "female",
    "വിധവ": "female",
    "विधवा": "female",
    "விதவை": "female",
    "mother": "female",
    "അമ്മ": "female",
    "grandmother": "female",
    "അമ്മൂമ്മ": "female",
    "father": "male",
    "അച്ഛൻ": "male",
    "grandfather": "male",
}

GRIEF_KEYWORDS = [
    "widow", "widower", "died", "death", "passed away",
    "വിധവ", "മരിച്ചു", "മരണം",
    "विधवा", "मर गया", "मृत्यु",
    "விதவை", "இறந்தார்",
    "feel alone", "nobody cares", "helpless",
]

AGE_PATTERNS = [
    r"(\d+)\s*(?:years?\s*old|yrs?|years)",
    r"(?:age|aged)\s*(\d+)",
    r"(\d+)\s*(?:വയസ്സ്|വയസ്)",
    r"(\d+)\s*(?:साल|वर्ष)",
    r"(\d+)\s*(?:வயது)",
]

_cached_prompt = None
_cached_client = None


def _load_prompt() -> dict:
    """Load prompt once, cache it."""
    global _cached_prompt
    if _cached_prompt is None:
        with open(PROMPT_PATH, "r", encoding="utf-8") as f:
            _cached_prompt = yaml.safe_load(f)
    return _cached_prompt


def _get_client() -> Groq:
    """Get or create the singleton Groq client."""
    global _cached_client
    if _cached_client is None:
        api_key = os.getenv("GROQ_API_KEY")
        if not api_key:
            raise RuntimeError("GROQ_API_KEY not set")
        _cached_client = Groq(api_key=api_key, timeout=30.0, max_retries=1)
    return _cached_client


def _postprocess(
    data: dict,
    raw_input: str,
    followup_answers: dict[str, str] | None = None,
) -> dict:
    """Fix common LLM mistakes using deterministic rules."""
    # 0. Fix None values for list fields
    if data.get("life_events") is None:
        data["life_events"] = []
    if data.get("missing_fields") is None:
        data["missing_fields"] = []

    # 1. Category: lowercase + validate
    cat = data.get("category")
    if isinstance(cat, str):
        cat = cat.strip().lower()
        if cat not in VALID_CATEGORIES:
            cat = None
    data["category"] = cat

    # 2. Gender: infer from keywords if missing
    if not data.get("gender"):
        lower = raw_input.lower()
        for keyword, gender in GENDER_KEYWORDS.items():
            if keyword in lower:
                data["gender"] = gender
                break

    # 3. Age: extract from the original message and follow-up answers
    if data.get("age") is None:
        followup_text = " ".join(
            f"{question}: {answer}"
            for question, answer in (followup_answers or {}).items()
            if isinstance(question, str) and isinstance(answer, str)
        )
        age_text = f"{raw_input}\n{followup_text}"

        for pattern in AGE_PATTERNS:
            match = re.search(pattern, age_text, re.IGNORECASE)
            if match:
                age = int(match.group(1))
                if 1 <= age <= 120:
                    data["age"] = age
                    break

    
    # 3b. State: recover a state explicitly given in follow-up answers
    if not data.get("state"):
        followup_text = " ".join(
            answer
            for answer in (followup_answers or {}).values()
            if isinstance(answer, str)
        ).strip()

        known_states = [
            "Andhra Pradesh", "Arunachal Pradesh", "Assam", "Bihar",
            "Chhattisgarh", "Goa", "Gujarat", "Haryana", "Himachal Pradesh",
            "Jharkhand", "Karnataka", "Kerala", "Madhya Pradesh",
            "Maharashtra", "Manipur", "Meghalaya", "Mizoram", "Nagaland",
            "Odisha", "Punjab", "Rajasthan", "Sikkim", "Tamil Nadu",
            "Telangana", "Tripura", "Uttar Pradesh", "Uttarakhand",
            "West Bengal", "Andaman and Nicobar Islands", "Chandigarh",
            "Dadra and Nagar Haveli and Daman and Diu", "Delhi",
            "Jammu and Kashmir", "Ladakh", "Lakshadweep", "Puducherry",
        ]

        for state in known_states:
            if re.search(
                rf"(?<!\w){re.escape(state)}(?!\w)",
                followup_text,
                re.IGNORECASE,
            ):
                data["state"] = state
                break



    # 4. Emotion: detect grief from keywords if LLM said neutral
    if data.get("emotion") == "neutral":
        lower = raw_input.lower()
        for keyword in GRIEF_KEYWORDS:
            if keyword.lower() in lower:
                data["emotion"] = "grief"
                break

    # 5. Life events: infer spouse_death from widow/widower
    if not data.get("life_events"):
        lower = raw_input.lower()
        if any(kw in lower for kw in ["widow", "വിധവ", "विधवा", "விதவை", "widower"]):
            data["life_events"] = ["spouse_death"]

    # 6. Remove inferred fields from missing_fields
    missing = data.get("missing_fields", [])
    for field in ["age", "gender", "category"]:
        if data.get(field) is not None and field in missing:
            missing.remove(field)
    data["missing_fields"] = missing

    # 7. Ensure raw_input is set
    data["raw_input"] = raw_input

    return data

@traceable(run_type="chain", name="1. Understanding Agent")
def understand(user_input: str, max_retries: int = 5, requested_mode: str | None = None,
               followup_answers: dict[str, str] | None = None) -> CitizenProfile:
    """Read a citizen's message and return a structured CitizenProfile."""
    prompt = _load_prompt()
    client = _get_client()
    user_message = prompt["user"].format(raw_input=user_input)

    if requested_mode:
        user_message += "\nSelected mode (preserve this): " + requested_mode
    if followup_answers:
        user_message += "\nFollow-up questions and citizen answers (JSON): " + json.dumps(followup_answers, ensure_ascii=False)

    last_error = None

    for attempt in range(max_retries):
        try:
            response = client.chat.completions.create(
                model=MODEL,
                messages=[
                    {"role": "system", "content": prompt["system"]},
                    {"role": "user", "content": user_message},
                ],
                temperature=0.1,
                response_format={"type": "json_object"},
            )

            raw_json = response.choices[0].message.content

            if not raw_json or not raw_json.strip():
                print(f"  [Understand] Empty response (attempt {attempt + 1}) — retrying")
                continue

            data = json.loads(raw_json)
            data = _postprocess(data, user_input, followup_answers)
            
            if requested_mode:
                data["mode"] = requested_mode
            if data.get("mode") == "mentor":
                has_direction = bool(data.get("interests") or data.get("strengths") or data.get("career_goal"))
                data["needs_followup"] = not has_direction and not data.get("user_said_enough", False)
            if data.get("mode") == "scheme" and data.get("state") and data.get("age") is not None and data.get("life_events"):
                # Enough context to discover possibilities; unknown eligibility facts remain checks.
                data["needs_followup"] = False
            if data.get("user_said_enough"):
                data["needs_followup"] = False
            if data.get("needs_followup") and not data.get("followup_questions"):
                questions = {
                    "en": "What subjects or activities do you enjoy, and what would you like to do next?",
                    "ml": "നിങ്ങൾക്ക് ഇഷ്ടമുള്ള വിഷയങ്ങളോ പ്രവർത്തനങ്ങളോ എന്തൊക്കെയാണ്? ഇനി എന്ത് ചെയ്യാനാണ് ആഗ്രഹം?",
                    "hi": "आपको कौन से विषय या गतिविधियाँ पसंद हैं, और आगे क्या करना चाहते हैं?",
                    "ta": "உங்களுக்கு எந்தப் பாடங்கள் அல்லது செயல்பாடுகள் பிடிக்கும்? அடுத்து என்ன செய்ய விரும்புகிறீர்கள்?",
                }
                if data.get("mode") == "scheme":
                    questions = {
                        "en": "Which state do you live in, and what is the applicant's age?",
                        "ml": "ഏത് സംസ്ഥാനത്താണ് താമസിക്കുന്നത്? അപേക്ഷകന്റെ പ്രായം എത്രയാണ്?",
                        "hi": "आप किस राज्य में रहते हैं और आवेदक की उम्र क्या है?",
                        "ta": "நீங்கள் எந்த மாநிலத்தில் வசிக்கிறீர்கள்? விண்ணப்பதாரரின் வயது என்ன?",
                    }
                data["followup_questions"] = [questions.get(data.get("language"), questions["en"])]
            if not data.get("needs_followup"):
                data["followup_questions"] = []

            try:
                return CitizenProfile(**data)
            except ValidationError as e:
                print(f"  [Understand] Schema validation failed (attempt {attempt + 1}): {e}")
                last_error = e
                continue

        except json.JSONDecodeError as e:
            print(f"  [Understand] Invalid JSON (attempt {attempt + 1}): {e}")
            last_error = e
            continue

        except RateLimitError as e:
            raise RuntimeError("The language service is busy or its quota has been reached. Please retry later.") from e

        except Exception as e:
            print(f"  [Understand] Groq error (attempt {attempt + 1}): {type(e).__name__}: {e}")
            last_error = e
            continue

    raise RuntimeError(
        f"Understanding failed after {max_retries} attempts. Last error: {last_error}"
    )

@router.post("/signup", response_model=UserResponse, status_code=201)
def signup(payload: UserSignup, db: Session = Depends(get_db)):
    """Register a new user."""
    existing = db.query(User).filter(User.email == payload.email).first()
    if existing:
        raise HTTPException(status_code=400, detail="Email already registered")

    user = User(
        email=payload.email,
        hashed_password=hash_password(payload.password),
        full_name=payload.full_name,
        role="citizen",
    )
    db.add(user)
    db.commit()
    db.refresh(user)
    return user


@router.post("/login", response_model=TokenResponse)
def login(payload: UserLogin, db: Session = Depends(get_db)):
    """Log in and receive a JWT token."""
    user = db.query(User).filter(User.email == payload.email).first()
    if not user or not verify_password(payload.password, user.hashed_password):
        raise HTTPException(status_code=401, detail="Invalid email or password")

    if not user.is_active:
        raise HTTPException(status_code=403, detail="User account is inactive")
    token = create_access_token(user.id, user.email, user.role)
    return TokenResponse(
        access_token=token,
        user=UserResponse.model_validate(user),
    )

"""FastAPI dependencies for authentication and authorization."""
from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from sqlalchemy.orm import Session
import jwt as pyjwt

from app.database import get_db
from app.models.user import User
from app.auth.jwt import decode_access_token


security = HTTPBearer()


def get_current_user(
    credentials: HTTPAuthorizationCredentials = Depends(security),
    db: Session = Depends(get_db),
) -> User:
    """Extract the current user from the JWT token."""
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        payload = decode_access_token(credentials.credentials)
        user_id = int(payload.get("sub"))
    except (pyjwt.PyJWTError, TypeError, ValueError):
        raise credentials_exception

    user = db.query(User).filter(User.id == user_id).first()
    if user is None:
        raise credentials_exception
    if not user.is_active:
        raise HTTPException(status_code=403, detail="User account is inactive")
    return user


def require_role(*allowed_roles: str):
    """Dependency factory: require the user to have one of the given roles."""
    def checker(current_user: User = Depends(get_current_user)) -> User:
        if current_user.role not in allowed_roles:
            raise HTTPException(
                status_code=403,
                detail=f"Access denied. Required roles: {allowed_roles}",
            )
        return current_user
    return checker

"""JWT token creation and verification."""
import os
from datetime import datetime, timedelta, timezone
from dotenv import load_dotenv
import jwt

load_dotenv()

SECRET_KEY = os.getenv("JWT_SECRET_KEY")
ALGORITHM = os.getenv("JWT_ALGORITHM", "HS256")
EXPIRY_HOURS = int(os.getenv("JWT_EXPIRY_HOURS", "24"))


def create_access_token(user_id: int, email: str, role: str) -> str:
    """Create a signed JWT token for a user."""
    payload = {
        "sub": str(user_id),
        "email": email,
        "role": role,
        "exp": datetime.now(timezone.utc) + timedelta(hours=EXPIRY_HOURS),
    }
    return jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)


def decode_access_token(token: str) -> dict:
    """Verify and decode a JWT token. Raises jwt.PyJWTError if invalid."""
    return jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])

"""Password hashing with bcrypt directly (no passlib)."""
import bcrypt


def hash_password(plain: str) -> str:
    """Convert a plain password into a bcrypt hash."""
    # bcrypt requires bytes, and has a 72-byte limit
    password_bytes = plain.encode("utf-8")
    if len(password_bytes) > 72:
        raise ValueError("Password must be at most 72 UTF-8 bytes")
    salt = bcrypt.gensalt(rounds=12)
    hashed = bcrypt.hashpw(password_bytes, salt)
    return hashed.decode("utf-8")


def verify_password(plain: str, hashed: str) -> bool:
    """Check if a plain password matches a stored hash."""
    try:
        password_bytes = plain.encode("utf-8")[:72]
        return bcrypt.checkpw(password_bytes, hashed.encode("utf-8"))
    except (ValueError, TypeError):
        return False

"""BM25 keyword search over the knowledge base chunks."""
import json
from pathlib import Path
from rank_bm25 import BM25Okapi


from app.rag.corpus import load_chunks


def load_all_chunks() -> list[dict]:
    return load_chunks()


class BM25Index:
    """In-memory BM25 index over chunks."""

    def __init__(self):
        self.chunks = load_all_chunks()
        tokenized = [(c["name"] + " " + c["text"]).lower().split() for c in self.chunks]
        self.bm25 = BM25Okapi(tokenized) if tokenized else None

    def search(self, query: str, top_k: int = 10, mode: str | None = None, state: str | None = None) -> list[dict]:
        """Return top_k chunks ranked by BM25 score."""
        if self.bm25 is None:
            return []
        tokens = query.lower().split()
        scores = self.bm25.get_scores(tokens)
        ranked = sorted(
            ((c, score) for c, score in zip(self.chunks, scores)
             if (mode is None or c["mode"] == mode)
             and (not state or c["state"] in (state.lower(), "central"))
             and score > 0),
            key=lambda x: x[1],
            reverse=True,
        )
        results = []
        for chunk, score in ranked[:top_k]:
            results.append({**chunk, "score": float(score)})
        return results


if __name__ == "__main__":
    index = BM25Index()
    print(f"Loaded {len(index.chunks)} chunks")
    results = index.search("My husband died what help can I get", top_k=3)
    for i, r in enumerate(results, 1):
        print(f"\n[{i}] {r['name']} (score: {r['score']:.4f})")
        print(f"    {r['text'][:150]}...")

# Module-level singleton â€” created once, reused forever
_index_instance = None


def get_bm25_index() -> BM25Index:
    """Get or create the singleton BM25 index."""
    global _index_instance
    if _index_instance is None:
        _index_instance = BM25Index()
    return _index_instance

"""Chunk clean text into overlapping pieces."""


def chunk_text(text: str, chunk_size: int = 2000, overlap: int = 200) -> list[str]:
    """
    Split text into chunks of approximately `chunk_size` characters
    with `overlap` characters of overlap between consecutive chunks.

    Args:
        text: clean text to split
        chunk_size: target characters per chunk
        overlap: characters shared between chunks

    Returns:
        List of chunk strings.
    """
    if len(text) <= chunk_size:
        return [text]

    chunks = []
    start = 0
    step = chunk_size - overlap

    while start < len(text):
        end = start + chunk_size
        chunk = text[start:end]

        # Try to break at sentence boundary (period + space)
        if end < len(text):
            last_period = chunk.rfind(". ")
            if last_period > chunk_size // 2:
                chunk = chunk[: last_period + 1]
                end = start + last_period + 1

        chunks.append(chunk.strip())
        start = end

    return [c for c in chunks if c]


if __name__ == "__main__":
    sample = "This is a sentence. " * 200

    print(f"Number of chunks: {len(chunks)}")
    print(f"First chunk length: {len(chunks[0])} chars")
    print(f"Last chunk length: {len(chunks[-1])} chars")

"""Clean extracted PDF text."""
import re


# Patterns for website navigation junk that gets captured when saving pages as PDF
JUNK_PATTERNS = [
    r"Check Eligibility",
    r"Sign in to apply",
    r"Sign In",
    r"Back",
    r"Enter scheme name to\s*search\.\.\.",
    r"Details\s*\n\s*Benefits\s*\n\s*\| \|",
    r"News and\s*\n?Updates",
    r"Share",
    r"No new news and\s*\n?updates available",
    r"Was this helpful\?",
    r"Frequently Asked\s*\n?Questions",
    r"Sources And\s*\n?References",
    r"Powered by\s*\n?Digital India Corporation",
    r"Ministry of Electronics & IT",
    r"Government of India\s*\n?Connect on Social",
    r"Quick Links\s+Useful Links\s+Get in touch",
    r"About Us\s*\n?Contact Us\s*\n?Screen Reader",
    r"Accessibility Statement\s*\n?Frequently Asked",
    r"Disclaimer\s*\n?Terms & Conditions\s*\n?Dashboard",
    r"Last Updated On",
    r"©\s*\d{4}",
    r"┬©\s*\d{4}",
    r"┬«",
]


def clean_text(text: str) -> str:
    """Remove common PDF artifacts and website navigation noise."""
    # Remove page numbers like "Page 1 of 10"
    text = re.sub(r"Page \d+ of \d+", "", text, flags=re.IGNORECASE)

    # Remove URL lines with timestamps
    text = re.sub(
        r"\d+/\d+/\d+,\s*\d+:\d+\s*(?:AM|PM)\s+.*?https?://\S+\s*\d+/\d+",
        "",
        text,
    )

    # Remove standalone URL lines
    text = re.sub(r"https?://\S+", "", text)

    # Remove the junk patterns
    for pattern in JUNK_PATTERNS:
        text = re.sub(pattern, "", text, flags=re.IGNORECASE)

    # Remove standalone numbers (page numbers, footnote markers)
    text = re.sub(r"^\s*\d+\s*$", "", text, flags=re.MULTILINE)

    # Collapse 3+ newlines into 2
    text = re.sub(r"\n{3,}", "\n\n", text)

    # Collapse multiple spaces into 1
    text = re.sub(r"[ \t]+", " ", text)

    # Remove leading/trailing whitespace on each line
    lines = [line.strip() for line in text.split("\n")]
    text = "\n".join(lines)

    # Remove empty lines at start/end
    return text.strip()


if __name__ == "__main__":

"""One metadata contract for sparse and dense retrieval."""
import json
from pathlib import Path

KB_DIR = Path(__file__).resolve().parents[2] / "data" / "knowledge_base"
CATEGORIES = ("central", "kerala", "general", "mentor")


def load_chunks(kb_dir: Path = KB_DIR) -> list[dict]:
    chunks = []
    for category in CATEGORIES:
        for path in sorted((kb_dir / category).glob("*.json")):
            doc = json.loads(path.read_text(encoding="utf-8"))
            meta = doc.get("metadata", {})
            if meta.get("active") is False:
                continue
            # Existing mentor JSONs have no mode tag. Their folder defines it.
            mode = "mentor" if category == "mentor" else "scheme"
            texts = list(doc.get("chunks", []))
            pages = list(doc.get("chunk_pages") or [0] * len(texts))
            names = [meta.get("name") or doc["doc_id"]] * len(texts)
            for reviewed in doc.get("reviewed_passages", []):
                texts.append(reviewed["text"])
                pages.append(reviewed["page"])
                names.append(reviewed["name"])
            for i, text in enumerate(texts):
                if not text.strip():
                    continue
                chunks.append({
                    "id": f"{doc['doc_id']}__chunk_{i}", "text": text,
                    "doc_id": doc["doc_id"], "name": names[i],
                    "category": meta.get("category") or "unknown",
                    "state": (meta.get("state") or (category if category in ("central", "kerala") else "")).lower(),
                    "mode": mode, "source_url": meta.get("source_url") or "",
                    "chunk_index": i,
                    "page": pages[i],
                    "edition": meta.get("edition", "Date not recorded"),
                    "retrieved_on": meta.get("retrieved_on", ""),
                })
    return chunks

"""Cross-encoder reranker for precision."""
import os
import torch
torch.set_num_threads(int(os.getenv("SAHAYAK_CPU_THREADS", "4")))

from sentence_transformers import CrossEncoder


_model = None
MODEL_NAME = "cross-encoder/ms-marco-MiniLM-L-6-v2"


def get_model() -> CrossEncoder:
    """Load the cross-encoder model lazily."""
    global _model
    if _model is None:
        print(f"Loading reranker model: {MODEL_NAME}")
        try:
            _model = CrossEncoder(MODEL_NAME, local_files_only=True)
        except OSError:
            _model = CrossEncoder(MODEL_NAME)
    return _model


def rerank(query: str, candidates: list[dict], text_key: str = "text", top_k: int = 5) -> list[dict]:
    """
    Re-score candidates by relevance to the query.

    Args:
        query: user query
        candidates: list of dicts, each with a text field
        text_key: which key holds the text
        top_k: how many to return

    Returns:
        Top_k candidates sorted by relevance score (descending).
    """
    if not candidates:
        return []

    model = get_model()
    pairs = [(query, c[text_key]) for c in candidates]
    scores = model.predict(pairs)

    ranked = sorted(
        zip(candidates, scores),
        key=lambda x: x[1],
        reverse=True,
    )
    results = []
    for cand, score in ranked[:top_k]:
        item = dict(cand)
        item["_rerank_score"] = float(score)
        results.append(item)
    return results

"""Reciprocal Rank Fusion (RRF) — merge multiple ranked lists."""


def reciprocal_rank_fusion(
    ranked_lists: list[list[dict]],
    k: int = 60,
    id_key: str = "id",
) -> list[dict]:
    """
    Merge multiple ranked lists using RRF.

    RRF score = sum over lists of 1 / (k + rank)

    Args:
        ranked_lists: list of ranked lists (each item must have id_key)
        k: RRF constant (default 60, from the original paper)
        id_key: which key identifies a unique item

    Returns:
        Merged list sorted by RRF score (descending).
    """
    scores = {}
    items_by_id = {}

    for ranked in ranked_lists:
        for rank, item in enumerate(ranked, start=1):
            item_id = item[id_key]
            items_by_id[item_id] = item
            scores[item_id] = scores.get(item_id, 0) + 1 / (k + rank)

    merged = []
    for item_id, score in sorted(scores.items(), key=lambda x: x[1], reverse=True):
        item = dict(items_by_id[item_id])
        item["_rrf_score"] = score
        merged.append(item)

    return merged


if __name__ == "__main__":
    list_a = [{"id": "x"}, {"id": "y"}, {"id": "z"}]
    list_b = [{"id": "y"}, {"id": "x"}, {"id": "w"}]

    merged = reciprocal_rank_fusion([list_a, list_b])
    print("Merged ranking:")
    for item in merged:
        print(f"  {item['id']}  rrf={item['_rrf_score']:.5f}")

"""ChromaDB wrapper for storing and searching embeddings."""
import os
from pathlib import Path

import chromadb
from dotenv import load_dotenv

load_dotenv()

CHROMA_PATH = os.getenv("CHROMA_PATH", "./chroma_db")
COLLECTION_NAME = "schemes"


def get_client():
    """Return a persistent ChromaDB client."""
    return chromadb.PersistentClient(path=CHROMA_PATH)


def get_or_create_collection():
    """Get the schemes collection, or create it if missing."""
    client = get_client()
    return client.get_or_create_collection(
        name=COLLECTION_NAME,
        metadata={"hnsw:space": "cosine"},
    )


def add_chunks(chunks: list[dict]) -> None:
    """
    Add chunks to the collection.

    Each chunk dict must have:
      - id: unique string
      - text: the chunk text
      - embedding: list of floats
      - metadata: dict of extra fields
    """
    collection = get_or_create_collection()
    collection.add(
        ids=[c["id"] for c in chunks],
        documents=[c["text"] for c in chunks],
        embeddings=[c["embedding"] for c in chunks],
        metadatas=[c["metadata"] for c in chunks],
    )


def search(query_embedding: list[float], n_results: int = 5, mode: str | None = None, state: str | None = None) -> dict:
    """Search for the most similar chunks to a query embedding."""
    collection = get_or_create_collection()
    if collection.count() == 0:
        return {"ids": [[]], "documents": [[]], "metadatas": [[]]}
    filters = []
    if mode:
        filters.append({"mode": mode})
    if state:
        filters.append({"state": {"$in": [state.lower(), "central"]}})
    where = filters[0] if len(filters) == 1 else {"$and": filters} if filters else None
    return collection.query(
        **({"where": where} if where else {}),
        query_embeddings=[query_embedding],
        n_results=min(n_results, collection.count()),
        include=["documents", "metadatas", "distances"],
    )


def count() -> int:
    """Return the number of chunks stored."""
    collection = get_or_create_collection()
    return collection.count()


def reset() -> None:
    """Delete the collection (useful for re-building the index)."""
    client = get_client()
    try:
        client.delete_collection(COLLECTION_NAME)
        print(f"Deleted collection: {COLLECTION_NAME}")
    except Exception:
        pass


if __name__ == "__main__":
    reset()
    print(f"Collection reset. Current count: {count()}")

"""Local OCR for printed English images. Images are not sent to an AI provider."""
from functools import lru_cache
from io import BytesIO
import numpy as np
from PIL import Image, UnidentifiedImageError

@lru_cache(maxsize=1)
def _engine():
    from rapidocr_onnxruntime import RapidOCR
    return RapidOCR(intra_op_num_threads=2, inter_op_num_threads=2)

def extract_text_from_image(image_bytes: bytes) -> str:
    if not image_bytes or len(image_bytes) > 10 * 1024 * 1024:
        return "OCR Error: Choose an image smaller than 10 MB."
    try:
        with Image.open(BytesIO(image_bytes)) as source:
            if source.width * source.height > 20_000_000:
                return "OCR Error: Choose an image below 20 megapixels."
            image = source.convert("RGB")
            image.thumbnail((2400, 2400))
            result, _ = _engine()(np.array(image))
        lines = [row[1] for row in (result or []) if row[2] >= .6]
        return "\n".join(lines) or "OCR Error: No clear printed text found. Try a sharper English document image."
    except (UnidentifiedImageError, OSError, ValueError):
        return "OCR Error: This image could not be read. Try a PNG or JPEG."
    except Exception:
        return "OCR Error: The local reader is unavailable. Check the OCR installation."

"""Voice input using Groq Whisper."""
import os
from groq import Groq
from dotenv import load_dotenv

load_dotenv()

def transcribe_audio(audio_bytes: bytes, filename: str = "audio.wav") -> str:
    """Transcribe audio using Groq's Whisper model."""
    if not audio_bytes or len(audio_bytes) > 24 * 1024 * 1024:
        return "Voice Error: Record a shorter message (under 24 MB)."
    api_key = os.getenv("GROQ_API_KEY")
    if not api_key:
        return "Error: GROQ_API_KEY not found."
        
    client = Groq(api_key=api_key, timeout=45, max_retries=1)
    
    try:
        transcription = client.audio.transcriptions.create(
            file=(filename, audio_bytes),
            model="whisper-large-v3",
            response_format="text",
            temperature=0.0
            # We removed the language parameter so it auto-detects Malayalam
        )
        
        # Handle different return types from Groq
        if hasattr(transcription, 'text'):
            return transcription.text
        elif isinstance(transcription, dict):
            return transcription.get('text', str(transcription))
        else:
            return str(transcription).strip()
            
    except Exception as e:
        return "Voice Error: Transcription failed. Please retry or type your message."

"""Web search fallback using DuckDuckGo (ddgs package)."""
import time
from ddgs import DDGS


def search(query: str, max_results: int = 5, retries: int = 2) -> list[dict]:
    """
    Search the web for a query. Returns list of results with title, snippet, url.

    Retries if the first attempt returns nothing (DuckDuckGo rate limits).
    """
    for attempt in range(retries + 1):
        try:
            results = []
            with DDGS() as ddgs:
                for r in ddgs.text(query, max_results=max_results):
                    results.append({
                        "title": r.get("title", ""),
                        "snippet": r.get("body", ""),
                        "url": r.get("href", ""),
                        "source": "web_search",
                    })
            if results:
                return results
            if attempt < retries:
                print(f"  [web_search] attempt {attempt + 1} returned 0 — retrying")
                time.sleep(2)
        except Exception as e:
            print(f"  [web_search] attempt {attempt + 1} failed: {e}")
            if attempt < retries:
                time.sleep(2)

    return []


if __name__ == "__main__":

def generate_html_report(profile, plan):
    t = labels(profile.language)
    e = lambda value: escape(str(value), quote=True)
    cards = []
    for index, option in enumerate(plan.eligible_schemes, 1):
        source = e(option.source_name)
        if option.source_url.startswith(("https://", "http://")):
            url = option.source_url + (f"#page={option.source_page}" if option.source_page and ".pdf" in option.source_url else "")
            source = f'<a href="{e(url)}" rel="noopener noreferrer">{source or e(t["source"])}</a>'
        else:
            source = e(t["missing"])
        cards.append(f"""<article class="card"><span class="number">{index:02d}</span>
          <h2>{e(option.name)}</h2><p>{e(option.benefits)}</p>
          <h3>{e(t['why'])}</h3><p>{e(option.why_relevant)}</p>
          <h3>{e(t['checks'])}</h3><p>{e(option.eligibility_note)}</p>
          <p class="muted">{e(t['deadline'])}: {e(option.deadline or t['unknown'])}</p>
          <details open><summary>{e(t['evidence'])}</summary><blockquote>{e(option.evidence_quote)}</blockquote>
          <p>{source} {(' · p. ' + str(option.source_page)) if option.source_page else ''}</p>
          <p class="muted">{e(option.source_edition)}</p></details></article>""")
    steps = ''.join(f'<li>{e(step)}</li>' for step in plan.next_steps)
    timeline = ''.join(f'<li><strong>{e(step.step)}</strong><p class="muted">{e(step.time_estimate)}</p></li>' for step in plan.timeline)
    documents = ''.join(f'<label class="check"><input type="checkbox"> <span>{e(item)}</span></label>' for item in plan.document_checklist)
    sections = ''
    if steps:
        sections += f'<section class="start"><h2>{e(t["next"])}</h2><ol>{steps}</ol></section>'
    sections += ''.join(cards)
    if timeline:
        sections += f'<section><h2>{e(t["timeline"])}</h2><ol class="roadmap">{timeline}</ol></section>'
    if documents:
        sections += f'<section><h2>{e(t["docs"])}</h2>{documents}</section>'
    return f"""<!doctype html><html lang="{e(profile.language)}"><head><meta charset="utf-8">

"""Accessible result cards; generated text is never interpolated into HTML."""
import hashlib
import streamlit as st
from app.ui.labels import labels
from app.ui.report import generate_html_report


def render_results(result):
    profile, plan, verdict = (result.get(k) for k in ("profile", "plan", "verdict"))
    if not profile or not plan:
        st.error("Please try again with a little more detail.")
        return
    text = labels(profile.language)
    if not verdict or not verdict.passed:
        st.warning(text["failed"])
        return
    if not plan.eligible_schemes and not plan.timeline:
        st.subheader(text["empty"])
        for step in plan.next_steps:
            st.info(step)
        return
    st.divider()
    st.subheader(text["title"])
    if plan.greeting:
        st.write(plan.greeting)
    st.caption(text["verify"])
    if plan.next_steps:
        with st.container(border=True):
            st.markdown("### " + text["next"])
            for step in plan.next_steps[:3]:
                st.write("→ " + step)
    st.markdown("### " + text[profile.mode if profile.mode in ("scheme", "mentor") else "scheme"])
    for index, option in enumerate(plan.eligible_schemes, 1):
        with st.container(border=True):
            st.markdown(f"#### {index}. {option.name}")
            st.write(option.benefits)
            if option.why_relevant:
                st.markdown("**" + text["why"] + "**")
                st.write(option.why_relevant)
            if option.eligibility_note:
                st.markdown("**" + text["checks"] + "**")
                st.write(option.eligibility_note)
            if option.deadline:
                st.caption(text["deadline"] + ": " + option.deadline)
            with st.expander(text["evidence"]):
                if option.evidence_quote:
                    st.write(option.evidence_quote)
                st.caption(option.source_name + (f" · p. {option.source_page}" if option.source_page else ""))
                st.caption(option.source_edition)
                if option.source_url.startswith(("https://", "http://")):
                    url = option.source_url + (f"#page={option.source_page}" if option.source_page and ".pdf" in option.source_url else "")
                    st.link_button(text["source"], url)
                else:
                    st.caption(text["missing"])
    if plan.timeline:
        st.markdown("### " + text["timeline"])
        for index, step in enumerate(plan.timeline, 1):
            st.write(f"{index}. {step.step}")
            if step.time_estimate:
                st.caption(step.time_estimate)
    if plan.document_checklist:
        st.markdown("### " + text["docs"])
        plan_key = hashlib.sha256(plan.model_dump_json().encode()).hexdigest()[:12]
        for index, document in enumerate(plan.document_checklist):
            st.checkbox(document, key=f"doc_{plan_key}_{index}")
    st.download_button(text["save"], generate_html_report(profile, plan),
                       file_name=f"sahayak_{profile.mode}.html", mime="text/html",
                       use_container_width=True)

def run_query(
    query: str,
    followup_answers: dict[str, str] | None = None,
):
    if not query.strip():
        st.warning("Please describe your situation.")
        return

    try:
        with st.spinner(
            "Understanding your request and preparing the next step..."
        ):
            result = run_pipeline(
                query,
                mode=st.session_state.page,
                followup_answers=followup_answers,
            )

    except Exception as exc:
        if (
            isinstance(exc, RuntimeError)
            and str(exc).startswith("The language service is busy")
        ):
            st.warning(
                "The language service is busy or its quota has been "
                "reached. Please retry later; your answers are still "
                "available."
            )
        else:
            st.error(
                "The request could not be completed. Please retry; "
                "your answers are still available."
            )
        return

    if result.get("needs_followup"):
        st.session_state.followup = {
            "questions": result.get("followup_questions", []),
            "original_query": query,
            "answers": followup_answers or {},
        }
        st.session_state.results = None
    else:
        st.session_state.followup = None
        st.session_state.results = result

    st.rerun()

"""The Sahayak LangGraph workflow: five agents in a stateful loop."""

import time

from langgraph.graph import StateGraph, START, END

from app.workflows.state import SahayakState
from app.workflows.nodes import (
    understand_node,
    research_node,
    plan_node,
    critic_node,
    replan_node,
)
from app.workflows.metrics import record_run


MAX_ITERATIONS = 3


def should_replan(state: SahayakState) -> str:
    """Decide whether to replan or finish after the critic runs."""
    verdict = state.get("verdict")
    iterations = state.get("iterations", 0)

    # An absent verdict means an upstream step did not produce one.
    if verdict is None:
        return "end"

    # A passing verdict means the plan is ready.
    if verdict.passed:
        return "end"

    # Return the best available plan when the iteration limit is reached.
    if iterations >= MAX_ITERATIONS:
        print(
            f"[Graph] Maximum iterations ({MAX_ITERATIONS}) reached. "
            "Returning the best available plan."
        )
        return "end"

    return "replan"


def build_graph():
    """Build and compile the Sahayak workflow."""
    graph = StateGraph(SahayakState)

    graph.add_node("understand", understand_node)
    graph.add_node("research", research_node)
    graph.add_node("plan", plan_node)
    graph.add_node("critic", critic_node)
    graph.add_node("replan", replan_node)

    graph.add_edge(START, "understand")

    graph.add_conditional_edges(
        "understand",
        lambda state: (
            "followup" if state.get("needs_followup") else "research"
        ),
        {
            "followup": END,
            "research": "research",
        },
    )

    graph.add_edge("research", "plan")
    graph.add_edge("plan", "critic")

    graph.add_conditional_edges(
        "critic",
        should_replan,
        {
            "replan": "replan",
            "end": END,
        },
    )

    # Recheck the revised plan with the critic.
    graph.add_edge("replan", "critic")

    return graph.compile()


_graph = None


def get_graph():
    """Return the cached compiled workflow."""
    global _graph

    if _graph is None:
        _graph = build_graph()

    return _graph


def run_pipeline(
    raw_input: str,
    mode: str | None = None,
    followup_answers: dict[str, str] | None = None,
) -> dict:
    """Run the workflow and record non-sensitive aggregate metrics."""
    if mode not in (None, "scheme", "mentor"):
        raise ValueError("Choose scheme or mentor mode")

    initial_state = {
        "raw_input": raw_input,
        "requested_mode": mode,
        "followup_answers": followup_answers or {},
    }

    graph = get_graph()
    start = time.perf_counter()

    try:
        result = graph.invoke(initial_state)
    except Exception:
        record_run({}, time.perf_counter() - start, failed=True)
        raise

    record_run(result, time.perf_counter() - start)
    return result


if __name__ == "__main__":
    sample = "My grandmother is 68, widow, lives in Kerala, no income"
    result = run_pipeline(sample)

    profile = result.get("profile")
    plan = result.get("plan")
    verdict = result.get("verdict")

    print("\n" + "=" * 60)
    print(f"Mode: {profile.mode if profile else 'N/A'}")
    print(f"Iterations: {result.get('iterations', 0)}")
    print(
        f"Verdict: "
        f"{'PASSED' if verdict and verdict.passed else 'FAILED'}"
    )
    print("=" * 60)

    if plan:
        print(f"\nSchemes: {len(plan.eligible_schemes)}")
        for scheme in plan.eligible_schemes:
            print(f"  - {scheme.name}")

        print(f"Estimated benefit: Rs {plan.estimated_annual_benefit:,}")

"""Non-sensitive run metrics; no prompts, profiles, answers or documents are recorded."""

import json
import os
from datetime import datetime, timezone
from pathlib import Path
from threading import Lock

_LOCK = Lock()

PATH = (
    Path(__file__).resolve().parents[2]
    / "data"
    / "eval"
    / "runtime_metrics.jsonl"
)


def record_run(result, seconds, failed=False):
    """Record aggregate run metadata without disrupting the pipeline."""
    if os.getenv("SAHAYAK_METRICS", "true").strip().lower() != "true":
        return

    try:
        if not isinstance(result, dict):
            result = {}

        profile = result.get("profile")
        research = result.get("research")
        verdict = result.get("verdict")

        row = {
            "timestamp": datetime.now(timezone.utc).isoformat(),
            "mode": getattr(profile, "mode", "unknown"),
            "language": getattr(profile, "language", "unknown"),
            "seconds": round(max(0.0, float(seconds)), 3),
            "error": bool(failed),
            "followup": bool(result.get("needs_followup")),
            "retrieved_chunks": getattr(research, "num_results", 0),
            "validation_passed": getattr(verdict, "passed", None),
            "iterations": result.get("iterations", 0),
        }

        with _LOCK:
            PATH.parent.mkdir(parents=True, exist_ok=True)
            with PATH.open("a", encoding="utf-8") as stream:
                stream.write(json.dumps(row) + "\n")

    except Exception:
        # Metrics must never prevent a citizen from receiving a result.
        pass

"""Node functions for the Sahayak LangGraph workflow.

Each node reads from SahayakState, calls an agent, and returns a dict of updates.
"""

from app.workflows.state import SahayakState
from app.agents.understanding import understand
from app.agents.research import research
from app.agents.planning import plan
from app.agents.critic import critique
from app.agents.replanner import replan


def understand_node(state: SahayakState) -> dict:
    """Extract a structured citizen profile from natural language."""
    raw_input = state.get("raw_input", "").strip()

    if not raw_input:
        raise ValueError("User input cannot be empty.")

    profile = understand(
        raw_input,
        requested_mode=state.get("requested_mode"),
        followup_answers=state.get("followup_answers") or {},
    )

    return {
        "profile": profile,
        "mode": profile.mode,
        "iterations": 0,
        "needs_followup": getattr(profile, "needs_followup", False),
        "followup_questions": getattr(profile, "followup_questions", []),
    }


def research_node(state: SahayakState) -> dict:
    """Retrieve relevant evidence for the citizen's profile."""
    profile = state.get("profile")

    if profile is None:
        return {"research": None}

    research_result = research(profile)
    return {"research": research_result}


def plan_node(state: SahayakState) -> dict:
    """Generate the initial action plan."""
    profile = state.get("profile")
    research_result = state.get("research")

    if profile is None or research_result is None:
        return {"plan": None}

    action_plan = plan(profile, research_result)
    return {"plan": action_plan}


def critic_node(state: SahayakState) -> dict:
    """Validate the plan using deterministic quality checks."""
    profile = state.get("profile")
    action_plan = state.get("plan")

    if profile is None or action_plan is None:
        return {
            "verdict": None,
            "iterations": state.get("iterations", 0),
        }

    verdict = critique(
        profile,
        action_plan,
        state.get("research"),
    )

    return {
        "verdict": verdict,
        "iterations": state.get("iterations", 0) + 1,
    }


def replan_node(state: SahayakState) -> dict:
    """Revise the plan using critic feedback and retrieved evidence."""
    profile = state.get("profile")
    action_plan = state.get("plan")
    verdict = state.get("verdict")
    research_result = state.get("research")

    if (
        profile is None
        or action_plan is None
        or verdict is None
        or verdict.passed
        or not verdict.errors
    ):
        return {}

    revised_plan = replan(
        profile,
        action_plan,
        verdict.errors,
        research=research_result,
    )

    return {"plan": revised_plan}

"""Build Chroma from the active corpus, reusing unchanged compatible embeddings."""
import sys
from pathlib import Path
sys.path.insert(0,str(Path(__file__).resolve().parents[1]))
from app.rag.corpus import load_chunks
BATCH_SIZE=32

def collect_chunks():
    return [{'id':c['id'],'text':c['text'],'metadata':{k:v for k,v in c.items() if k not in ('id','text')}} for c in load_chunks()]

def main():
    from app.rag.embed import embed_batch, MODEL_NAME
    from app.rag.vector_store import add_chunks, reset, count, get_or_create_collection
    chunks=collect_chunks()
    if not chunks:
        raise SystemExit('No active chunks. Import sources first; existing index was preserved.')
    collection=get_or_create_collection()
    reusable={}
    if (collection.metadata or {}).get('embedding_model')==MODEL_NAME:
        old=collection.get(include=['documents','embeddings'])
        reusable={ident:(text,vector) for ident,text,vector in zip(old['ids'],old['documents'],old['embeddings'])}
    missing=[]
    for chunk in chunks:
        saved=reusable.get(chunk['id'])
        if saved is not None and saved[0]==chunk['text']:
            chunk['embedding']=list(saved[1])
        else:
            missing.append(chunk)
    print(f"Active passages: {len(chunks)}; new embeddings: {len(missing)}",flush=True)
    if missing:
        for chunk,vector in zip(missing,embed_batch([c['text'] for c in missing])):
            chunk['embedding']=vector
    # Finish all model work before replacing the index. Restart running apps after this command.
    reset()
    for i in range(0,len(chunks),BATCH_SIZE):
        add_chunks(chunks[i:i+BATCH_SIZE])
    get_or_create_collection().modify(metadata={'embedding_model':MODEL_NAME})
    print(f'Index ready: {count()} passages',flush=True)
if __name__=='__main__': main()

"""Evaluate RAG retrieval quality against a golden dataset."""
import sys
import json
import time
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

GOLDEN_PATH = Path("data/eval/golden_queries.json")


def load_golden() -> list[dict]:
    """Load the golden query dataset."""
    with open(GOLDEN_PATH, "r", encoding="utf-8") as f:
        return json.load(f)["queries"]


def run_retrieval(query: str, top_k: int = 5) -> list[str]:
    """Run hybrid retrieval, return list of doc_ids in ranked order."""
    from app.rag.embed import embed_text
    from app.rag.vector_store import search as vector_search
    from app.rag.bm25 import get_bm25_index
    from app.rag.rrf import reciprocal_rank_fusion
    from app.rag.rerank import rerank

    # Vector search
    query_vec = embed_text(query)
    vec_results = vector_search(query_vec, n_results=15, mode="scheme")
    vec_hits = []
    for i in range(len(vec_results["ids"][0])):
        meta = vec_results["metadatas"][0][i]
        vec_hits.append({
            "id": vec_results["ids"][0][i],
            "text": vec_results["documents"][0][i],
            "doc_id": meta.get("doc_id", ""),
            "name": meta.get("name", ""),
        })

    # BM25
    bm25 = get_bm25_index()
    bm25_hits = bm25.search(query, top_k=15, mode="scheme")

    # RRF
    fused = reciprocal_rank_fusion([vec_hits, bm25_hits], id_key="id")

    # Rerank
    reranked = rerank(query, fused[:20], text_key="text", top_k=top_k)

    # Extract doc_ids in ranked order (dedupe by doc)
    doc_ids = []
    seen = set()
    for r in reranked:
        doc_id = r.get("doc_id", "")
        if doc_id and doc_id not in seen:
            seen.add(doc_id)
            doc_ids.append(doc_id)
    return doc_ids


def evaluate() -> dict:
    """Run evaluation across all golden queries."""
    queries = load_golden()
    results = []

    for item in queries:
        qid = item["id"]
        query = item["query"]
        expected = item["expected_docs"]

        ranked = run_retrieval(query, top_k=5)

        # Find rank of first expected doc
        rank = None
        for i, doc_id in enumerate(ranked):
            if doc_id in expected:
                rank = i + 1  # 1-based
                break

        results.append({
            "id": qid,
            "query": query,
            "expected": expected,
            "retrieved": ranked,
            "rank": rank,
        })

    # Calculate metrics
    total = len(results)
    top1 = sum(1 for r in results if r["rank"] == 1)
    top3 = sum(1 for r in results if r["rank"] and r["rank"] <= 3)
    top5 = sum(1 for r in results if r["rank"] and r["rank"] <= 5)

    # MRR (Mean Reciprocal Rank)
    mrr = sum(1 / r["rank"] if r["rank"] else 0 for r in results) / total

    # Precision@3 (fraction of top 3 that are expected)
    prec_at_3_sum = 0
    for r in results:
        top_3 = r["retrieved"][:3]
        relevant = sum(1 for d in top_3 if d in r["expected"])
        prec_at_3_sum += relevant / 3 if top_3 else 0
    precision_at_3 = prec_at_3_sum / total

    return {
        "total": total,
        "top1": top1,
        "top3": top3,
        "top5": top5,
        "top1_pct": round(100 * top1 / total, 2),
        "top3_pct": round(100 * top3 / total, 2),
        "top5_pct": round(100 * top5 / total, 2),
        "mrr": round(mrr, 4),
        "precision_at_3": round(precision_at_3, 4),
        "results": results,
    }


def print_report(metrics: dict, save_path: str = None) -> None:
    """Print a clean report and optionally save as JSON."""
    print("=" * 70)
    print("RAG RETRIEVAL EVALUATION")
    print("=" * 70)
    print(f"\nTotal queries: {metrics['total']}")
    print(f"Top-1 accuracy: {metrics['top1']}/{metrics['total']} "
          f"({metrics['top1_pct']}%)")
    print(f"Top-3 accuracy: {metrics['top3']}/{metrics['total']} "
          f"({metrics['top3_pct']}%)")
    print(f"Top-5 accuracy: {metrics['top5']}/{metrics['total']} "
          f"({metrics['top5_pct']}%)")
    print(f"MRR: {metrics['mrr']}")
    print(f"Precision@3: {metrics['precision_at_3']}")

    print("\n" + "-" * 70)
    print("PER-QUERY BREAKDOWN")
    print("-" * 70)
    for r in metrics["results"]:
        mark = "OK " if r["rank"] == 1 else ("-- " if r["rank"] is None else "..")
        rank_str = f"#{r['rank']}" if r["rank"] else "MISS"
        print(f"  [{mark}] {r['id']}  {rank_str:6s}  {r['query'][:50]}")

    if save_path:
        with open(save_path, "w", encoding="utf-8") as f:
            json.dump(metrics, f, indent=2)
        print(f"\nFull results saved to {save_path}")


if __name__ == "__main__":
    start = time.time()
    metrics = evaluate()
    elapsed = time.time() - start

    print_report(metrics, save_path="data/eval/baseline_results.json")
    print(f"\nTotal time: {elapsed:.1f}s")

"""Import downloaded official PDFs with page provenance and extraction checks."""
import hashlib
from html import unescape
import json
import re
from datetime import date
from pathlib import Path
from pypdf import PdfReader

ROOT = Path(__file__).resolve().parents[1]


def pieces(text, size=1200, overlap=160):
    text = re.sub(r"\s+", " ", text).strip()
    start = 0
    while start < len(text):
        end = min(start + size, len(text))
        if end < len(text):
            boundary = text.rfind(" ", start + size // 2, end)
            if boundary > start:
                end = boundary
        yield text[start:end]
        if end == len(text):
            break
        start = max(start + 1, end - overlap)


def main():
    audit = []
    for entry in json.loads((ROOT / "data/source_manifest.json").read_text(encoding="utf8")):
        path = ROOT / entry["path"]
        row = {"id": entry["id"], "source_url": entry["url"]}
        if not path.exists():
            audit.append({**row, "status": "download_unavailable"})
            continue
        if entry.get("extraction_status") == "needs_ocr":
            audit.append({**row, "status": "needs_ocr"})
            continue
        try:
            target = ROOT / "data/knowledge_base" / entry["folder"] / (entry["id"] + ".json")
            with path.open("rb") as stream:
                digest = hashlib.file_digest(stream, "sha256").hexdigest()
            if target.exists():
                existing = json.loads(target.read_text(encoding="utf8"))
                if existing.get("metadata", {}).get("sha256") == digest and (entry.get("format") != "html" or existing.get("metadata", {}).get("extraction_version") == 2):
                    existing["metadata"]["source_url"] = entry["url"]
                    target.write_text(json.dumps(existing, ensure_ascii=False, indent=2), encoding="utf8")
                    audit.append({**row, "status": "text_extracted", "chunks": len(existing["chunks"])})
                    continue
            is_html = entry.get("format") == "html"
            if is_html:
                text = path.read_text(encoding="utf8")
                text = re.sub(r"<(script|style)\b[^>]*>.*?</\1>", " ", text, flags=re.S|re.I)
                text = re.sub(r"\s+", " ", unescape(re.sub(r"<[^>]+>", " ", text))).strip()
                end = text.find(entry.get("section_end", "History of Pension"))
                start = text.rfind(entry.get("section_start", "Criteria for Allotting"), 0, end)
                if start < 0 or end <= start:
                    raise ValueError("Expected pension criteria section missing")
                pages = [text[start:end].split("* Special Amount")[0]]
            else:
                reader = PdfReader(path)
                pages = [(p.extract_text() or "").replace("\x00", "") for p in reader.pages]
            useful = [(i + 1, t) for i, t in enumerate(pages) if len(t.strip()) >= 100]
            if len(useful) < max(1, len(pages) * .25):
                audit.append({**row, "status": "needs_ocr", "pages": len(pages)})
                continue
            chunks, page_ids = [], []
            for page, text in useful:
                for part in pieces(text):
                    if len(part) >= 100:
                        chunks.append(part)
                        page_ids.append(0 if is_html else page)
            with path.open("rb") as stream:
                digest = hashlib.file_digest(stream, "sha256").hexdigest()
            doc = {"doc_id": entry["id"], "source_file": entry["path"],
                   "metadata": {"name": entry["name"], "source_url": entry["url"],
                                "publisher": entry["publisher"], "edition": entry["edition"],
                                "retrieved_on": date.today().isoformat(), "sha256": digest,
                                "state": entry.get("state", "central"), "active": True, "extraction_version": 2},
                   "chunks": chunks, "chunk_pages": page_ids, "num_chunks": len(chunks),
                   "char_count": sum(map(len, chunks))}
            target = ROOT / "data/knowledge_base" / entry["folder"] / (entry["id"] + ".json")
            target.parent.mkdir(parents=True, exist_ok=True)
            target.write_text(json.dumps(doc, ensure_ascii=False, indent=2), encoding="utf8")
            audit.append({**row, "status": "text_extracted", "pages": len(pages), "chunks": len(chunks)})
            print(entry["id"], len(chunks), "passages", flush=True)
        except Exception as exc:
            audit.append({**row, "status": "extraction_failed", "error_type": type(exc).__name__})
    target = ROOT / "data/eval/source_audit.json"
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text(json.dumps(audit, indent=2), encoding="utf8")
    print(json.dumps(audit, indent=2))


if __name__ == "__main__":
    main()

"""Print the Sahayak project tree with file sizes and counts."""
import os
from pathlib import Path


# Folders/files to skip (noise)
SKIP_DIRS = {
    "venv", ".venv", "__pycache__", ".git", ".cache",
    "chroma_db", ".pytest_cache", ".mypy_cache",
    "node_modules", ".idea", ".vscode",
}

SKIP_FILES = {
    ".DS_Store", "Thumbs.db",
}


def human_size(n: int) -> str:
    for unit in ["B", "KB", "MB", "GB"]:
        if n < 1024:
            return f"{n:.1f} {unit}"
        n /= 1024
    return f"{n:.1f} TB"


def count_files(path: Path) -> tuple[int, int]:
    """Return (file_count, total_size) for a directory."""
    files = 0
    total = 0
    for root, dirs, filenames in os.walk(path):
        dirs[:] = [d for d in dirs if d not in SKIP_DIRS]
        for f in filenames:
            if f in SKIP_FILES:
                continue
            files += 1
            try:
                total += (Path(root) / f).stat().st_size
            except OSError:
                pass
    return files, total


def walk(path: Path, prefix: str = "", is_last: bool = True):
    """Recursively print the tree."""
    entries = sorted(
        [e for e in path.iterdir()
         if e.name not in SKIP_DIRS and e.name not in SKIP_FILES],
        key=lambda e: (not e.is_dir(), e.name.lower()),
    )

    for i, entry in enumerate(entries):
        last = i == len(entries) - 1
        connector = "└── " if last else "├── "

        if entry.is_dir():
            file_count, total_size = count_files(entry)
            print(f"{prefix}{connector}{entry.name}/  "
                  f"[{file_count} files, {human_size(total_size)}]")
            extension = "    " if last else "│   "
            walk(entry, prefix + extension, last)
        else:
            try:
                size = human_size(entry.stat().st_size)
            except OSError:
                size = "?"
            print(f"{prefix}{connector}{entry.name}  ({size})")


def main():
    root = Path(".")
    print(f"\n{'='*70}")
    print(f"Sahayak — Project Tree")
    print(f"Root: {root.resolve()}")
    print(f"{'='*70}\n")
    print(f"{root.name}/")
    walk(root)

    # Summary
    total_files, total_size = count_files(root)
    print(f"\n{'='*70}")
    print(f"Total: {total_files} files, {human_size(total_size)}")
    print(f"{'='*70}\n")


if __name__ == "__main__":
    main()

"""Test the full pipeline: Plan → Critic → Replan → Critic, max 3 loops."""
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

from app.agents.understanding import understand
from app.agents.research import research
from app.agents.planning import plan as build_plan
from app.agents.critic import critique
from app.agents.replanner import replan


MAX_ITERATIONS = 3


def run_full_pipeline(text: str) -> dict:
    """Run the full 5-agent loop."""
    print(f"\n{'='*70}")
    print(f"INPUT: {text}")
    print(f"{'='*70}")

    # Step 1: Understanding
    profile = understand(text)
    print(f"Mode: {profile.mode}, Age: {profile.age}, Gender: {profile.gender}")

    # Step 2: Research
    research_result = research(profile)
    print(f"Research: {research_result.num_results} chunks (source: {research_result.source})")

    # Step 3: Initial plan
    action_plan = build_plan(profile, research_result)
    print(f"Initial plan: {len(action_plan.eligible_schemes)} schemes")

    # Step 4-5: Critic + Replanner loop
    iteration = 0
    final_verdict = None

    while iteration < MAX_ITERATIONS:
        iteration += 1
        verdict = critique(profile, action_plan)
        print(f"\n  Iteration {iteration}: {'PASSED' if verdict.passed else 'FAILED'}")

        if verdict.errors:
            for e in verdict.errors:
                print(f"    Error: {e}")

        if verdict.passed:
            final_verdict = verdict
            break

        if iteration >= MAX_ITERATIONS:
            print(f"    Max iterations reached — returning best-effort plan")
            final_verdict = verdict
            break

        # Replan
        print(f"    Replanning...")
        action_plan = replan(profile, action_plan, verdict.errors)

    # Final report
    print(f"\n{'='*70}")
    print(f"FINAL RESULT (after {iteration} iteration(s))")
    print(f"{'='*70}")
    print(f"Status: {'PASSED' if final_verdict.passed else 'FAILED (best-effort)'}")
    print(f"Schemes: {len(action_plan.eligible_schemes)}")
    for s in action_plan.eligible_schemes:
        print(f"  - {s.name}")
    print(f"Documents: {len(action_plan.document_checklist)}")
    print(f"Timeline steps: {len(action_plan.timeline)}")
    print(f"Est. annual benefit: Rs {action_plan.estimated_annual_benefit:,}")

    if final_verdict.warnings:
        print(f"\nWarnings:")
        for w in final_verdict.warnings:
            print(f"  - {w}")

    return {
        "profile": profile,
        "research": research_result,
        "plan": action_plan,
        "verdict": final_verdict,
        "iterations": iteration,
    }


def main():
    tests = [
        "My grandmother is 68, widow, lives in Kerala, no income",
        "I'm 17, finished 12th, OBC category, need scholarship",
    ]
    for text in tests:
        run_full_pipeline(text)


if __name__ == "__main__":
    main()

def hybrid_search(query: str, top_k: int = 5, verbose: bool = True):
    """Run vector + BM25, fuse with RRF, rerank."""
    # 1. Vector search
    query_vec = embed_text(query)
    vec_results = vector_search(query_vec, n_results=10)
    vec_hits = []

        vec_hits.append({

            "state": vec_results["metadatas"][0][i].get("state", ""),

    fused = reciprocal_rank_fusion([vec_hits, bm25_hits], id_key="id")

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

from app.agents.understanding import understand
from app.agents.research import research


TESTS = [
    ("Widow pension",
     "എന്റെ അമ്മൂമ്മ 68 വയസ്സ്, വിധവ, കേരളത്തിൽ"),
    ("OBC scholarship",
     "I'm 17 and finished 12th, I need OBC scholarship"),
    ("Old age pension",
     "My father is 65 and retired, what pension can he get?"),
]


def main():
    start = time.time()

    for label, text in TESTS:
        print(f"\n{'='*70}")
        print(f"TEST: {label}")
        print(f"INPUT: {text}")
        print(f"{'='*70}")

        profile = understand(text)
        print(f"Mode: {profile.mode}")
        print(f"Age: {profile.age}")
        print(f"Gender: {profile.gender}")
        print(f"Category: {profile.category}")

        result = research(profile)
        print(f"Query: {result.query}")
        print(f"Found: {result.num_results} chunks")
        print()

        for i, c in enumerate(result.chunks, 1):
            print(f"  [{i}] {c.name}")
            print(f"      score: {c.rerank_score:.3f}")
            print(f"      text: {c.text[:100]}...")
            print()

    print(f"\nTotal time: {time.time() - start:.2f}s")


if __name__ == "__main__":
    main()

    return {"profile": CitizenProfile(mode="mentor"),
            "plan": ActionPlan(greeting="Your next step", next_steps=["Compare courses"],

def setup_retrieval(monkeypatch, score=2):
    vector = Mock(return_value={'ids': [[]], 'documents': [[]], 'metadatas': [[]]})
    monkeypatch.setitem(sys.modules, 'app.rag.embed', SimpleNamespace(embed_text=lambda q: [0.0]))
    monkeypatch.setitem(sys.modules, 'app.rag.vector_store', SimpleNamespace(search=vector))
    monkeypatch.setitem(sys.modules, 'app.rag.rerank', SimpleNamespace(
        rerank=lambda query, candidates, **kw: [{**c, '_rerank_score': score} for c in candidates]))
    web = Mock(side_effect=AssertionError('Mentor must not use unverified web snippets'))
    monkeypatch.setitem(sys.modules, 'app.tools.web_search', SimpleNamespace(search=web))
    return vector, web


def test_mentor_research_works_with_stale_dense_index(monkeypatch):
    vector, web = setup_retrieval(monkeypatch)
    result = agent.research(CitizenProfile(mode='mentor', state='kerala',
        raw_input='I finished 12th, I am confused.', interests=['biology'], career_goal='doctor NEET'))
    assert result.chunks
    assert any(c.doc_id == 'nta_neet_2026' for c in result.chunks)
    assert any(len(c.text) > 500 for c in result.chunks)
    assert vector.call_args.kwargs['mode'] == 'mentor'
    assert vector.call_args.kwargs['state'] is None
    web.assert_not_called()
    assert all(c.source_url.startswith('https://') and c.page > 0 for c in result.chunks)


def test_irrelevant_mentor_results_stay_empty(monkeypatch):
    setup_retrieval(monkeypatch, score=-10)
    result = agent.research(CitizenProfile(mode='mentor', raw_input='biology'))
    assert result.chunks == []
    plan = planning.plan(CitizenProfile(mode='mentor'), result)
    assert 'panchayat' not in ' '.join(plan.next_steps).lower()
    assert not plan.eligible_schemes


def test_empty_mentor_message_is_localized():
    research = ResearchResult(query='career', mode='mentor')
    for language in ('en', 'ml', 'hi', 'ta'):
        result = planning.plan(CitizenProfile(mode='mentor', language=language), research)
        assert result.next_steps and 'panchayat' not in result.next_steps[0].lower()


def test_vector_filters_are_applied_before_ranking(monkeypatch):
    import importlib
    monkeypatch.setitem(sys.modules, 'chromadb', SimpleNamespace())
    module = importlib.import_module('app.rag.vector_store')
    collection = Mock()
    collection.count.return_value = 100
    monkeypatch.setattr(module, 'get_or_create_collection', lambda: collection)
    module.search([0.0], mode='mentor')
    assert collection.query.call_args.kwargs['where'] == {'mode': 'mentor'}
    module.search([0.0], mode='scheme', state='Kerala')
    assert collection.query.call_args.kwargs['where'] == {'$and': [
        {'mode': 'scheme'}, {'state': {'$in': ['kerala', 'central']}}]}

def test_entrance_bulletins_respect_explicit_career_intent():
    assert not agent._source_matches_intent('nta_neet_2026','Engineering after class 12 physics mathematics')
    assert not agent._source_matches_intent('nta_jee_2026','How to become a doctor through MBBS?')
    assert agent._source_matches_intent('nta_neet_2026','Compare engineering and medical careers')
    assert agent._source_matches_intent('nta_jee_2026','Compare engineering and medical careers')
    assert agent._source_matches_intent('nta_neet_2026','What can I study after school?')
